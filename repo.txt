```
services:
  user-db:
    image: postgres:16-alpine
    container_name: messenger-user-db
    environment:
      POSTGRES_USER: ${USERS_DB_USER}
      POSTGRES_PASSWORD: ${USERS_DB_PASSWORD}
      POSTGRES_DB: ${USERS_DB}
    ports:
      - "5431:5432"
    volumes:
      - user_db_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}"]
      interval: 5s
      timeout: 5s
      retries: 5
    networks:
      - messenger-net

  chat-db:
    image: postgres:16-alpine
    container_name: messenger-chat-db
    environment:
      POSTGRES_USER: ${CHAT_DB_USER}
      POSTGRES_PASSWORD: ${CHAT_DB_PASSWORD}
      POSTGRES_DB: ${CHAT_DB}
    ports:
      - "5432:5432"
    volumes:
      - chat_db_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U $${POSTGRES_USER} -d $${POSTGRES_DB}"]
      interval: 5s
      timeout: 5s
      retries: 5
    networks:
      - messenger-net
  
  user-db-migrations:
    image: migrate/migrate
    container_name: messenger-user-db-migrations
    volumes:
      - ./apps/user-service/migrations:/migrations
    command: -path=/migrations -database="${USERS_DB_URL}" up
    depends_on:
      user-db:
        condition: service_healthy
    networks:
      - messenger-net
  chat-db-migrations:
    image: migrate/migrate
    container_name: messenger-chat-db-migrations
    volumes:
      - ./apps/chat-service/migrations:/migrations
    command: -path=/migrations -database="${CHAT_DB_URL}" up
    depends_on:
      chat-db:
        condition: service_healthy
    networks:
      - messenger-net
      
  redis-db:
    image: redis:7-alpine
    container_name: messenger-redis-db
    ports:
      - "6379:6379"
    volumes:
      - redisdata:/data
    networks:
      - messenger-net

  nginx:
    image: nginx:alpine
    container_name: messenger-nginx
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx.conf:/etc/nginx/conf.d/default.conf:ro
    depends_on:
      - user-service
      - chat-service
      - ws-gateway
    networks:
      - messenger-net
  user-service:
    build:
      context: .
      dockerfile: apps/user-service/Dockerfile
    container_name: messenger-user-service
    environment:
      DATABASE_URL: ${USERS_DB_URL}
      REDIS_HOST: "redis-db"
      REDIS_PORT: "6379"
    depends_on:
      user-db-migrations:
        condition: service_completed_successfully
      redis-db:
        condition: service_started
    networks:
      - messenger-net

  chat-service:
    build:
      context: .
      dockerfile: apps/chat-service/Dockerfile
    container_name: messenger-chat-service
    environment:
      DATABASE_URL: ${CHAT_DB_URL}
      REDIS_HOST: "redis-db"
      REDIS_PORT: "6379"
    depends_on:
      chat-db-migrations:

        condition: service_started

      - messenger-net

      - messenger-net
```

## File: nginx.conf

```
server {
    listen 80;
    server_name localhost;


    location = /internal/auth/verify {
        # internal запрещает клиентам вызывать этот эндпоинт напрямую снаружи
        internal; 
        # Проксируем подзапрос проверки сессии в user-service
        proxy_pass http://user-service:8080/api/v1/auth/verify;
        
        # Нам не нужно тело оригинального запроса для проверки сессии
        proxy_method GET;
        proxy_pass_request_body off;
        proxy_set_header Content-Length "";
        # Передаем оригинальные куки и заголовки авторизации
        proxy_set_header X-Original-URI $request_uri;
        proxy_set_header Cookie $http_cookie;
        proxy_set_header Authorization $http_authorization;
    }

    location /api/v1/users {
        proxy_pass http://user-service:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    location /api/v1/users/search {
        auth_request /internal/auth/verify;
        auth_request_set $user_id $upstream_http_x_user_id;
        proxy_set_header X-User-ID $user_id;
        proxy_pass http://user-service:8080;
        
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
    location /api/v1/users/me {
        auth_request /internal/auth/verify;
        auth_request_set $user_id $upstream_http_x_user_id;
        proxy_set_header X-User-ID $user_id;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass http://user-service:8080;
    }

    location /api/v1/users/logout {
        auth_request /internal/auth/verify;
        auth_request_set $user_id $upstream_http_x_user_id;
        proxy_set_header X-User-ID $user_id;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_pass http://user-service:8080;
    }

    location /api/v1/chats {
        auth_request /internal/auth/verify;
        auth_request_set $user_id $upstream_http_x_user_id;
        proxy_set_header X-User-ID $user_id;
        proxy_pass http://chat-service:8080;
    }

    location /api/v1/ws {
```

```
package main

import (
	"context"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"sync"
	"syscall"
	"time"

	"chat-service/internal/config"
	handler "chat-service/internal/delivery/http"
	"chat-service/internal/delivery/http/middleware"
	repository "chat-service/internal/repository/postgres"
	"chat-service/internal/usecase"
	"messenger/pkg/pubsub"

	_ "github.com/jackc/pgx/v5/stdlib"
	"github.com/jmoiron/sqlx"
	"github.com/redis/go-redis/v9"
)

func main() {
	logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
	slog.SetDefault(logger)

	cfg := config.MustLoad()

	// 1. Подключение к PostgreSQL
	var db *sqlx.DB
	var err error

	maxRetries := 10
	for i := 1; i <= maxRetries; i++ {
		db, err = sqlx.Connect("pgx", cfg.DatabaseURL)
		if err == nil {
			logger.Info("successfully connected to database")
			break
		}

		logger.Warn("database connection failed, retrying...",
			slog.Int("attempt", i),
			slog.Int("max_attempts", maxRetries),
			slog.String("error", err.Error()),
		)
		time.Sleep(2 * time.Second)
	}

	if err != nil {
		logger.Error("failed to connect to database after max retries", slog.String("error", err.Error()))
		panic(err)
	}
	defer db.Close()

	// 2. Инициализация Redis Pub/Sub
	rdb := redis.NewClient(&redis.Options{
		Addr: cfg.Redis.Addr(),
	})

	if err := rdb.Ping(context.Background()).Err(); err != nil {
		logger.Error("failed to connect to redis", slog.String("error", err.Error()))
		panic(err)
	}
	defer rdb.Close()

	ps := pubsub.New(rdb)

	// 3. Сборка UseCase и HTTP Handlers
	storage := repository.NewPostgresStorage(db)
	chatService := usecase.NewChatService(storage, ps)
	h := handler.NewHandler(chatService, logger)

	mux := http.NewServeMux()

	mux.HandleFunc("POST /api/v1/chats/{id}/messages", h.HandleSendMessage)
	mux.HandleFunc("GET /api/v1/chats/{id}/messages", h.HandleGetMessagesPaginated)
	mux.HandleFunc("POST /api/v1/chats", h.HandleCreateChat)
	mux.HandleFunc("POST /api/v1/chats/{id}", h.HandleJoinToChat)
	mux.HandleFunc("PUT /api/v1/chats/{id}/read", h.HandleUpdateLastReadMessageID)
	mux.HandleFunc("GET /api/v1/chats", h.HandleGetUserChats)

	wrappedMux := middleware.Logger(mux)
	server := &http.Server{
		Addr:    ":" + cfg.ServerPort,
		Handler: wrappedMux,
	}

	fatalErr := make(chan error, 1)
	var wg sync.WaitGroup

	wg.Add(1)
	go func() {
		defer wg.Done()
		logger.Info("starting HTTP server", slog.String("port", cfg.ServerPort))
		if err := server.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			logger.Error("HTTP server failed", slog.String("err", err.Error()))
			fatalErr <- err
		}
	}()

	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)

	select {
	case sig := <-quit:
		logger.Info("shutdown signal received", slog.String("signal", sig.String()))
	case err := <-fatalErr:
		logger.Error("fatal error, initiating shutdown", slog.String("err", err.Error()))
	}

	logger.Info("stopping services...")

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	if err := server.Shutdown(ctx); err != nil {
		logger.Error("HTTP server forced shutdown", slog.String("err", err.Error()))
	}

	wg.Wait()
	logger.Info("all services stopped")
}
```

```
package handler

import (
	"chat-service/internal/delivery/http/dto"
	customErrors "chat-service/internal/errors"
	"encoding/json"
	"errors"
	"log/slog"
	"net/http"

	"github.com/google/uuid"
)

func (h *Handler) HandleCreateChat(w http.ResponseWriter, r *http.Request) {
	w.Header().Set("Content-Type", "application/json")
	userIDStr := r.Header.Get("X-User-ID")
	senderID, err := uuid.Parse(userIDStr)
	if err != nil {
		w.WriteHeader(http.StatusUnauthorized)
		return
	}
	var req dto.CreateChatRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		w.WriteHeader(http.StatusBadRequest)
		return
	}
	chatID, err := h.ChatService.CreateChat(r.Context(), req.ChatType, req.Title, senderID, req.UserIDs)
	resp := dto.CreateChatResponse{ChatID: chatID}
	if err != nil {
		if errors.Is(err, customErrors.DirectChatExist) {
			h.Logger.Info("direct chat already exists")
			w.WriteHeader(http.StatusConflict)
			json.NewEncoder(w).Encode(resp)
			return
		}
		if errors.Is(err, customErrors.IncorrectChatType) {
			h.Logger.Info("incorrect chat type")
			w.WriteHeader(http.StatusBadRequest)
			return
		}
		h.Logger.Error("failed to create chat", slog.String("error", err.Error()))
		w.WriteHeader(http.StatusInternalServerError)
		return
	}
	w.WriteHeader(http.StatusCreated)
	json.NewEncoder(w).Encode(resp)
}
```

```
package middleware

import (
	"log/slog"
	"net/http"
	"time"
)

// Специальная структура-обертка, которая "шпионит" за статус-кодом
type responseWriterWrapper struct {
	http.ResponseWriter
	statusCode int
}

// Переопределяем метод WriteHeader, чтобы запомнить статус-код
func (rw *responseWriterWrapper) WriteHeader(code int) {
	rw.statusCode = code
	rw.ResponseWriter.WriteHeader(code)
}

// Logger возвращает мидлварь, которая логирует информацию о каждом HTTP-запросе
func Logger(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		// 1. Фиксируем время начала запроса
		start := time.Now()

		// 2. Передаем управление следующему хендлеру в цепочке (нашему бизнес-коду)
		// Оборачиваем оригинальный ResponseWriter, ставим дефолтный статус 200
		wrappedWriter := &responseWriterWrapper{ResponseWriter: w, statusCode: http.StatusOK}

		// Передаем обертку дальше
		next.ServeHTTP(wrappedWriter, r)

		// 3. После того как хендлер отработал, считаем время выполнения
		duration := time.Since(start)

		// 4. Пишем красивый структурированный лог
		slog.Info("HTTP request",
			slog.String("method", r.Method),
			slog.String("path", r.URL.Path),
			slog.Int("status", wrappedWriter.statusCode),
			slog.Duration("duration", duration),
		)
	})
}
```

```
func (s *PostgresStorage) CreateChat(ctx context.Context, title *string, chatType string, userIDs []uuid.UUID) (uuid.UUID, error) {

	if err != nil {

	}
	defer tx.Rollback()
	queryCreate := `INSERT INTO chats (title, type) VALUES ($1, $2) RETURNING id`

	}
	for _, u := range userIDs {

		if _, err := tx.ExecContext(ctx, queryJoin, chatID, u); err != nil {

		}
	}

	return chatID, tx.Commit()
}
```

```
	LEFT JOIN LATERAL 
	(SELECT m.content, m.created_at, m.updated_at

	WHERE m.chat_id = cm.chat_id 
	ORDER BY m.created_at DESC 
	LIMIT 1) last_msg ON true

	JOIN LATERAL

	FROM messages m

	WHERE cm.user_id = $1
	ORDER BY last_msg.created_at DESC NULLS LAST`

	}
```

```
func (s *PostgresStorage) GetMessagesPaginated(ctx context.Context, chatID uuid.UUID, lastSeenID *uuid.UUID, limit int) ([]entity.Message, error) {
	var query string
	var args []any

	if lastSeenID == nil {
		// Первичная загрузка чата (самые свежие сообщения)
		query = `
			SELECT id, chat_id, sender_id, content, created_at 
			FROM messages 
			WHERE chat_id = $1 AND is_deleted = FALSE 
			ORDER BY id DESC LIMIT $2`
		args = []any{chatID, limit}
	} else {
		// Подгрузка старых сообщений при скролле вверх (id < last_seen_id)
		query = `
			SELECT id, chat_id, sender_id, content, created_at 
			FROM messages 
			WHERE chat_id = $1 AND id < $2 AND is_deleted = FALSE 
			ORDER BY id DESC LIMIT $3`
		args = []any{chatID, *lastSeenID, limit}
	}
	var messages []pgmodel.Message
	err := s.DB.SelectContext(ctx, &messages, query, args...)
	if err != nil {
		return nil, err
	}
	entityMessages := make([]entity.Message, len(messages))
	for i := range messages {
		entityMessages[i] = messages[i].ToEntity()
	}
	return entityMessages, nil
}
```

```
func (s *ChatService) CreateChat(ctx context.Context, chatType string, title *string, userID uuid.UUID, userIDs []uuid.UUID) (uuid.UUID, error) {
	userIDs = append(userIDs, userID)
	switch {
	case chatType == "direct" && len(userIDs) == 2:

		if err != nil {

		}
		if chatID != uuid.Nil {

		}
		return s.repo.CreateChat(ctx, nil, chatType, userIDs)
	case chatType == "group":
		return s.repo.CreateChat(ctx, title, chatType, userIDs)
	default:
		return uuid.Nil, customErrors.IncorrectChatType
	}
}

```
```
package usecase

import (
	"chat-service/internal/entity"
	"context"
	"fmt"
	"messenger/pkg/pubsub"
	"time"

	"github.com/google/uuid"
)

func ToPayload(m *entity.Message) pubsub.MessagePayload {
	return pubsub.MessagePayload{
		ID:        m.ID,
		ChatID:    m.ChatID,
		SenderID:  m.SenderID,
		Content:   m.Content,
		CreatedAt: m.CreatedAt,
	}
}
func (s *ChatService) SendMessage(ctx context.Context, chatID, senderID uuid.UUID, content string) (*entity.Message, error) {
	// 1. Генерируем UUIDv7 (или используем uuid.NewV7())
	msgID, err := uuid.NewV7()
	if err != nil {
		return nil, err
	}

	msg := &entity.Message{
		ID:        msgID,
		ChatID:    chatID,
		SenderID:  senderID,
		Content:   content,
		CreatedAt: time.Now(),
	}

	// 2. Атомарно сохраняем в БД
	if err := s.repo.SaveMessage(ctx, msg); err != nil {
		return nil, err
	}

	// 3. Публикуем событие в Redis Pub/Sub за O(1)
	// Вся логика фильтрации и разогрева WebSocket переходит на сторону ws-gateway
	event := pubsub.ChatEvent{
		Type:      pubsub.EventNewMessage,
		ChatID:    chatID.String(),
		SenderID:  senderID.String(),
		Payload:   ToPayload(msg),
		CreatedAt: msg.CreatedAt,
	}

	channel := fmt.Sprintf("chat:%s", chatID.String())
	if err := s.ps.Publish(ctx, channel, event); err != nil {
		return msg, fmt.Errorf("message saved, but failed to publish event: %w", err)
	}

	return msg, nil
}
```

```
package usecase

import (
	"context"
	"fmt"
	"messenger/pkg/pubsub"
	"time"

	"github.com/google/uuid"
)

func (s *ChatService) UpdateLastReadMessage(ctx context.Context, chatID, userID, messageID uuid.UUID) error {
	err := s.repo.UpdateLastReadMessage(ctx, chatID, userID, messageID)
	if err != nil {
		return err
	}

	// 2. Публикуем событие прочтения в Pub/Sub за O(1)
	event := pubsub.ChatEvent{
		Type:     pubsub.EventMessageRead,
		ChatID:   chatID.String(),
		SenderID: userID.String(), // В данном контексте — кто прочитал
		Payload: map[string]string{
			"message_id": messageID.String(),
		},
		CreatedAt: time.Now(),
	}

	channel := fmt.Sprintf("chat:%s", chatID.String())
	if err := s.ps.Publish(ctx, channel, event); err != nil {
		// Ошибка публикации не должна отменять факт прочтения в БД
		return fmt.Errorf("read status updated, but failed to publish event: %w", err)
	}

	return nil
}
```

```
package main

import (
	"context"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"
	"user-service/internal/config"
	httpHandler "user-service/internal/delivery/http"
	"user-service/internal/delivery/http/middleware"
	repository "user-service/internal/repository/postgres"
	cache "user-service/internal/repository/redis"
	"user-service/internal/usecase"

	"github.com/jmoiron/sqlx"
	"github.com/redis/go-redis/v9"
)

func main() {
	logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
	slog.SetDefault(logger)

	cfg := config.MustLoad()

	var db *sqlx.DB
	var err error

	maxRetries := 10
	for i := 1; i <= maxRetries; i++ {
		db, err = sqlx.Connect("pgx", cfg.DatabaseURL)
		if err == nil {
			logger.Info("successfully connected to database")
			break
		}

		logger.Warn("database connection failed, retrying...",
			slog.Int("attempt", i),
			slog.Int("max_attempts", maxRetries),
			slog.String("error", err.Error()),
		)
		time.Sleep(2 * time.Second)
	}

	if err != nil {
		logger.Error("failed to connect to database after max retries", slog.String("error", err.Error()))
		panic(err)
	}
	defer db.Close()

	opt := &redis.Options{
		Addr:     cfg.Redis.Addr(),
		Password: cfg.Redis.Password,
		DB:       cfg.Redis.DB,
	}
	client := redis.NewClient(opt)

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	if err := client.Ping(ctx).Err(); err != nil {
		logger.Error("redis connection failed", slog.String("error", err.Error()))
		panic(err)
	}
	cache := cache.NewRedisRepository(client)
	storage := repository.NewPostgresStorage(db)
	userUsecase := usecase.NewUsecase(cache, storage)
	userHandler := httpHandler.NewUserHandler(userUsecase, logger)

	mux := http.NewServeMux()

	mux.HandleFunc("GET /api/v1/users/me", userHandler.GetUserHandler)
	mux.HandleFunc("POST /api/v1/users/login", userHandler.LoginUserHandler)
	mux.HandleFunc("POST /api/v1/users/logout", userHandler.LogoutUserHandler)
	mux.HandleFunc("POST /api/v1/users/register", userHandler.RegisterUserHandler)
	mux.HandleFunc("GET /api/v1/auth/verify", userHandler.CheckSessionIDHandler)
	mux.HandleFunc("GET /api/v1/users/search", userHandler.SearchUsersHandler)
	wrappedMux := middleware.Logger(mux)
	server := &http.Server{
		Addr:    ":" + cfg.ServerPort,
		Handler: wrappedMux,
	}
	// Канал для аварийных ошибок — буфер 2, чтобы обе горутины могли записать не блокируясь
	fatalErr := make(chan error, 1)
	// Запускаем HTTP-сервер
	go func() {
		logger.Info("starting HTTP server", slog.String("port", cfg.ServerPort))
		if err := server.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			logger.Error("HTTP server failed", slog.String("err", err.Error()))
			fatalErr <- err // сообщаем об аварии
		}
	}()
	// Ждём сигнала остановки (Ctrl+C или docker stop)
	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)

	// Ждём: либо сигнал от ОС, либо авария одного из сервисов
	select {
	case sig := <-quit:
		logger.Info("shutdown signal received", slog.String("signal", sig.String()))
	case err := <-fatalErr:
		logger.Error("fatal error, initiating shutdown", slog.String("err", err.Error()))
	}

	// Дальше — тот же код остановки
	logger.Info("stopping services...")

	ctx, cancel = context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	if err := server.Shutdown(ctx); err != nil {
		logger.Error("HTTP server forced shutdown", slog.String("err", err.Error()))
	}

	logger.Info("all services stopped")
}
```

```
	Port     string `env:"REDIS_PORT" env-default:"6379"`
	Password string `env:"REDIS_PASSWORD"`
	DB       int    `env:"REDIS_DB" env-default:"0"`
}

// Addr собирает host:port для redis.Options — так проще, чем городить URL
func (r RedisConfig) Addr() string {
	return fmt.Sprintf("%s:%s", r.Host, r.Port)
}

func MustLoad() *Config {
	var cfg Config

	// 1. Сначала читаем .env (если он есть)
	_ = cleanenv.ReadConfig(".env", &cfg)

	// 2. Поверх перезаписываем реальными переменными окружения из Docker/OS
	if err := cleanenv.ReadEnv(&cfg); err != nil {
		slog.Error("failed to load env config", slog.String("error", err.Error()))
		panic(err)
	}

	return &cfg
}
```

```
func (h *UserHandler) CheckSessionIDHandler(w http.ResponseWriter, r *http.Request) {
	r.Header.Del("X-User-ID")

	if err != nil {
		w.WriteHeader(http.StatusUnauthorized)
		return
	}
	userSession, err := h.UC.ValidateSession(r.Context(), cookie.Value)
	if err != nil {
		h.Logger.Error("error checking session id", slog.String("error", err.Error()))
		w.WriteHeader(http.StatusInternalServerError)
		return
	}
	if !userSession.IsValid {
		w.WriteHeader(http.StatusUnauthorized)
		return
	}
	w.Header().Add("X-User-ID", userSession.UserID.String())
	w.WriteHeader(http.StatusOK)

}
```

```
package httpHandler

import (
	"encoding/json"
	"errors"
	"log/slog"
	"net/http"
	"user-service/internal/delivery/http/dto"
	customErrors "user-service/internal/errors"
)

func (h *UserHandler) LoginUserHandler(w http.ResponseWriter, r *http.Request) {
	var req dto.LoginRequest
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		h.Logger.Warn("failed to decode request body", slog.String("error", err.Error()))
		w.WriteHeader(http.StatusBadRequest)
		return
	}
	loginResult, err := h.UC.LoginUser(r.Context(), req.Email, req.Password)
	if err != nil {
		if errors.Is(err, customErrors.ErrInvalidEmailOrPassword) {
			w.WriteHeader(http.StatusBadRequest)
			json.NewEncoder(w).Encode(dto.ErrorResponse{
				Error: "Неверная почта или пароль",
			})
			return
		} else {
			h.Logger.Error("failed to login user", slog.String("error", err.Error()))
			w.WriteHeader(http.StatusInternalServerError)
			return
		}
	}
	http.SetCookie(w, &http.Cookie{
		Name:     "session_token",
		Value:    loginResult.SessionToken,
		Path:     "/",
		MaxAge:   86400,
		HttpOnly: true,
		Secure:   false,
		SameSite: http.SameSiteLaxMode,
	})
	w.WriteHeader(http.StatusOK)
}
func (h *UserHandler) LogoutUserHandler(w http.ResponseWriter, r *http.Request) {
	cookie, err := r.Cookie("session_token")
	if err != nil {
		w.WriteHeader(http.StatusUnauthorized)
		return
	}
	if err := h.UC.DeleteSession(r.Context(), cookie.Value); err != nil {
		h.Logger.Error("failed to delete session", slog.String("error", err.Error()))
		w.WriteHeader(http.StatusInternalServerError)
		return
	}
	http.SetCookie(w, &http.Cookie{
		Name:     "session_token",
		Value:    "",
		Path:     "/",
		MaxAge:   -1,
		HttpOnly: true,
		Secure:   false,
		SameSite: http.SameSiteLaxMode,
	})

	w.WriteHeader(http.StatusNoContent)
}
```

```
package httpHandler

import (
	"encoding/json"
	"log/slog"
	"net/http"
	"strconv"
	"user-service/internal/delivery/http/dto"

	"github.com/google/uuid"
)

func (h *UserHandler) SearchUsersHandler(w http.ResponseWriter, r *http.Request) {
	userIDStr := r.Header.Get("X-User-ID")
	userID, err := uuid.Parse(userIDStr)
	if err != nil {
		w.WriteHeader(http.StatusUnauthorized)
		return
	}
	queryParams := r.URL.Query()
	query := queryParams.Get("query")
	limitStr := queryParams.Get("limit")
	cursorStr := queryParams.Get("cursor")

	limit := 10
	if limitStr != "" {
		if parsedLimit, err := strconv.Atoi(limitStr); err == nil && parsedLimit > 0 {
			limit = parsedLimit
		}
	}

	var cursor *string
	if cursorStr != "" {
		cursor = &cursorStr
	}
	users, err := h.UC.SearchUsers(r.Context(), query, cursor, limit, userID)
	if err != nil {
		h.Logger.Error("failed to search users", slog.String("error", err.Error()))
		w.WriteHeader(http.StatusInternalServerError)
		return
	}
	searchResult := make([]dto.SearchUsersResponse, 0, limit)
	for _, u := range users {
		searchResult = append(searchResult, *dto.ToDTOSearchResult(&u))
	}
	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusOK)
	json.NewEncoder(w).Encode(searchResult)
}
```

```
package middleware

import (
	"log/slog"
	"net/http"
	"time"
)

// Специальная структура-обертка, которая "шпионит" за статус-кодом
type responseWriterWrapper struct {
	http.ResponseWriter
	statusCode int
}

// Переопределяем метод WriteHeader, чтобы запомнить статус-код
func (rw *responseWriterWrapper) WriteHeader(code int) {
	rw.statusCode = code
	rw.ResponseWriter.WriteHeader(code)
}

// Logger возвращает мидлварь, которая логирует информацию о каждом HTTP-запросе
func Logger(next http.Handler) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		// 1. Фиксируем время начала запроса
		start := time.Now()

		// 2. Передаем управление следующему хендлеру в цепочке (нашему бизнес-коду)
		// Оборачиваем оригинальный ResponseWriter, ставим дефолтный статус 200
		wrappedWriter := &responseWriterWrapper{ResponseWriter: w, statusCode: http.StatusOK}

		// Передаем обертку дальше
		next.ServeHTTP(wrappedWriter, r)

		// 3. После того как хендлер отработал, считаем время выполнения
		duration := time.Since(start)

		// 4. Пишем красивый структурированный лог
		slog.Info("HTTP request",
			slog.String("method", r.Method),
			slog.String("path", r.URL.Path),
			slog.Int("status", wrappedWriter.statusCode),
			slog.Duration("duration", duration),
		)
	})
}
```

```
func (s *PostgresStorage) UserExists(ctx context.Context, email, username string) (bool, error) {
	var exists bool

	query := `SELECT EXISTS(SELECT 1 FROM users WHERE email = $1 OR username = $2)`

	err := s.DB.GetContext(ctx, &exists, query, email, username)
	if err != nil {
		return false, err
	}
```

```
package usecase

import (
	"context"
	"crypto/rand"
	"crypto/sha256"
	"encoding/hex"
	"errors"
	"time"
	"user-service/internal/entity"
	customErrors "user-service/internal/errors"

	"github.com/google/uuid"
)

func (uc *UsecaseImpl) ValidateSession(ctx context.Context, sessionID string) (*entity.UserSession, error) {
	userID, err := uc.AuthRepo.GetUserID(ctx, hashToken(sessionID))
	if err != nil {
		if errors.Is(err, customErrors.ErrSessionInvalid) {
			return &entity.UserSession{IsValid: false}, nil
		}
		return nil, err
	}
	return &entity.UserSession{
		UserID:  userID,
		IsValid: true,
	}, nil
}
func generateSessionToken() (rawToken string, tokenHash string, err error) {
	raw := make([]byte, 32)
	if _, err := rand.Read(raw); err != nil {
		return "", "", err
	}
	rawToken = hex.EncodeToString(raw)
	tokenHash = hashToken(rawToken)
	return rawToken, tokenHash, nil
}

func hashToken(token string) string {
	sum := sha256.Sum256([]byte(token))
	return hex.EncodeToString(sum[:])
}
func (uc *UsecaseImpl) CreateSession(ctx context.Context, userID uuid.UUID) (string, error) {
	rawToken, tokenHash, err := generateSessionToken()
	if err != nil {
		return "", err
	}
	if err := uc.AuthRepo.SaveSessionToken(ctx, tokenHash, userID, 24*time.Hour); err != nil {
		return "", err
	}
	return rawToken, err
}

func (uc *UsecaseImpl) DeleteSession(ctx context.Context, sessionToken string) error {
	return uc.AuthRepo.DeleteSessionToken(ctx, sessionToken)
}
```

```
package main

import (
	"context"
	"log/slog"
	"net/http"
	"os"
	"os/signal"
	"syscall"
	"time"

	"messenger/pkg/pubsub"
	"ws-gateway/internal/config"
	"ws-gateway/internal/delivery/ws"
	"ws-gateway/internal/hub"
	gwpubsub "ws-gateway/internal/pubsub"

	"github.com/redis/go-redis/v9"
)

func main() {
	logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
	slog.SetDefault(logger)

	cfg := config.MustLoad()

	// 1. Подключение к Redis
	rdb := redis.NewClient(&redis.Options{
		Addr: cfg.Redis.Addr(),
	})

	if err := rdb.Ping(context.Background()).Err(); err != nil {
		logger.Error("failed to connect to redis", slog.String("error", err.Error()))
		panic(err)
	}
	defer rdb.Close()

	ps := pubsub.New(rdb)

	// 2. Создание менеджера подписок Redis
	manager := gwpubsub.NewDynamicPubSubManager(ps)

	// 3. Создание Hub с привязкой коллбэков подписки/отписки
	h := hub.New(
		manager.SubscribeChat,   // Вызывается при появлении первого юзера чата на этом инстансе
		manager.UnsubscribeChat, // Вызывается при уходе последнего юзера чата с этого инстанса
	)
	manager.SetHub(h)

	// 4. HTTP / WebSocket сервер
	wsHandler := ws.NewHandler(h, ps)
	mux := http.NewServeMux()
	mux.HandleFunc("GET /api/v1/ws", wsHandler.HandleConnect)

	httpServer := &http.Server{
		Addr:    ":" + cfg.ServerPort,
		Handler: mux,
	}

	fatalErr := make(chan error, 1)

	go func() {
		logger.Info("starting HTTP (ws) server", slog.String("port", cfg.ServerPort))
		if err := httpServer.ListenAndServe(); err != nil && err != http.ErrServerClosed {
			logger.Error("HTTP server failed", slog.String("err", err.Error()))
			fatalErr <- err
		}
	}()

	quit := make(chan os.Signal, 1)
	signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)

	select {
	case sig := <-quit:
		logger.Info("shutdown signal received", slog.String("signal", sig.String()))
	case err := <-fatalErr:
		logger.Error("fatal error, initiating shutdown", slog.String("err", err.Error()))
	}

	logger.Info("stopping services...")

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	if err := httpServer.Shutdown(ctx); err != nil {
		logger.Error("HTTP server forced shutdown", slog.String("err", err.Error()))
	}

	logger.Info("all services stopped")
}
```

```
package ws

import (
	"context"
	"encoding/json"
	"fmt"
	"log/slog"
	"messenger/pkg/pubsub"
	"net/http"
	"strings"
	"time"

	"ws-gateway/internal/hub"

	"github.com/google/uuid"
	"github.com/gorilla/websocket"
)

var upgrader = websocket.Upgrader{
	CheckOrigin: func(r *http.Request) bool { return true },
}

const (
	pongWait   = 60 * time.Second
	pingPeriod = (pongWait * 9) / 10
	writeWait  = 10 * time.Second
)

type Handler struct {
	hub *hub.Hub
	ps  *pubsub.PubSub
}

func NewHandler(h *hub.Hub, ps *pubsub.PubSub) *Handler {
	return &Handler{hub: h, ps: ps}
}

func (h *Handler) HandleConnect(w http.ResponseWriter, r *http.Request) {
	userIDStr := r.Header.Get("X-User-ID")
	userID, err := uuid.Parse(userIDStr)
	if err != nil {
		w.WriteHeader(http.StatusUnauthorized)
		return
	}

	var activeChatIDs []uuid.UUID
	if chatIDsParam := r.URL.Query().Get("chat_ids"); chatIDsParam != "" {
		rawIDs := strings.Split(chatIDsParam, ",")
		for _, rawID := range rawIDs {
			if id, err := uuid.Parse(strings.TrimSpace(rawID)); err == nil {
				activeChatIDs = append(activeChatIDs, id)
			}
		}
	}

	conn, err := upgrader.Upgrade(w, r, nil)
	if err != nil {
		slog.Error("ws upgrade failed", slog.String("error", err.Error()))
		return
	}

	client := h.hub.Register(userID, conn, activeChatIDs)
	slog.Info("user connected",
		slog.String("user_id", userID.String()),
		slog.Int("active_chats_count", len(activeChatIDs)),
	)

	go h.writePump(client)
	go h.readPump(client)
}

func (h *Handler) writePump(client *hub.Client) {
	ticker := time.NewTicker(pingPeriod)
	defer func() {
		ticker.Stop()
		client.Conn().Close()
	}()

	for {
		select {
		case data, ok := <-client.Send():
			client.Conn().SetWriteDeadline(time.Now().Add(writeWait))
			if !ok {

				client.Conn().WriteMessage(websocket.CloseMessage, []byte{})
				return
			}
			if err := client.Conn().WriteMessage(websocket.TextMessage, data); err != nil {
				return
			}

		case <-ticker.C:
			client.Conn().SetWriteDeadline(time.Now().Add(writeWait))
			if err := client.Conn().WriteMessage(websocket.PingMessage, nil); err != nil {
				return
			}
		}
	}
}

func (h *Handler) readPump(client *hub.Client) {
	defer func() {
		h.hub.Unregister(client)
		slog.Info("user disconnected", slog.String("user_id", client.UserID().String()))
	}()

	conn := client.Conn()
	conn.SetReadDeadline(time.Now().Add(pongWait))
	conn.SetPongHandler(func(string) error {
		conn.SetReadDeadline(time.Now().Add(pongWait))
		return nil
	})

	for {
		_, data, err := conn.ReadMessage()
		if err != nil {
			slog.Warn("failed to read message with ws", slog.String("error", err.Error()))
			return
		}

		var eventType pubsub.TypeEvent
		if err := json.Unmarshal(data, &eventType); err != nil {
			slog.Warn("failed to unmarshal data to event type", slog.String("error", err.Error()))
			continue
		}

		switch eventType.Type {
		case pubsub.EventUserTyping:
			h.handleTyping(client, data)
		}
	}
}

func (h *Handler) handleTyping(client *hub.Client, data []byte) {
	var event pubsub.ChatEvent
	event.CreatedAt = time.Now()
	if err := json.Unmarshal(data, &event); err != nil {
		slog.Warn("failed to unmarshal data to userTypingEvent", slog.String("error", err.Error()))
		return
	}

	chatID, err := uuid.Parse(event.ChatID)
	if err != nil {
		slog.Warn("failed to parse chat_id to uuid from string", slog.String("error", err.Error()))
		return
	}

	if !h.hub.IsUserInChat(client.UserID(), chatID) {
		slog.Warn("user hasn't chat with id", slog.String("chat_id", event.ChatID))
		return
	}

	event.SenderID = client.UserID().String()
	channel := fmt.Sprintf("chat:%s", event.ChatID)
	if err := h.ps.Publish(context.Background(), channel, event); err != nil {
		slog.Error("failed to publish event", slog.String("error", err.Error()))
	}
}
```

```
package hub

import (
	"encoding/json"
	"log/slog"
	"sync"

	"github.com/google/uuid"
	"github.com/gorilla/websocket"
)

const sendBufferSize = 32

type Client struct {
	conn   *websocket.Conn
	userID uuid.UUID
	send   chan []byte
}

type Hub struct {
	mu        sync.RWMutex
	clients   map[uuid.UUID]*Client
	userChats map[uuid.UUID]map[uuid.UUID]bool
	chatUsers map[uuid.UUID]map[uuid.UUID]bool

	OnFirstChatUserJoined func(chatID uuid.UUID)
	OnLastChatUserLeft    func(chatID uuid.UUID)
}

func New(onSub, onUnsub func(chatID uuid.UUID)) *Hub {
	return &Hub{
		clients:               make(map[uuid.UUID]*Client),
		userChats:             make(map[uuid.UUID]map[uuid.UUID]bool),
		chatUsers:             make(map[uuid.UUID]map[uuid.UUID]bool),
		OnFirstChatUserJoined: onSub,
		OnLastChatUserLeft:    onUnsub,
	}
}
func (c *Client) Conn() *websocket.Conn { return c.conn }
func (c *Client) Send() chan []byte     { return c.send }
func (c *Client) UserID() uuid.UUID     { return c.userID }
func (h *Hub) IsUserInChat(userID, chatID uuid.UUID) bool {
	h.mu.RLock()
	defer h.mu.RUnlock()
	return h.userChats[userID][chatID]
}

// Register создаёт нового Client и возвращает его — вызывающий код (HandleConnect)
// использует возвращённый *Client, чтобы запустить его readPump/writePump.
func (h *Hub) Register(userID uuid.UUID, conn *websocket.Conn, activeChatIDs []uuid.UUID) *Client {
	h.mu.Lock()

	if old, ok := h.clients[userID]; ok {
		old.conn.Close() // старое соединение больше не нужно — новое его заменяет
	}

	client := &Client{
		conn:   conn,
		userID: userID,
		send:   make(chan []byte, sendBufferSize),
	}
	h.clients[userID] = client

	if _, exists := h.userChats[userID]; !exists {
		h.userChats[userID] = make(map[uuid.UUID]bool)
	}

	chatsToSubscribe := make([]uuid.UUID, 0)
	for _, chatID := range activeChatIDs {
		h.userChats[userID][chatID] = true
		if _, exists := h.chatUsers[chatID]; !exists {
			h.chatUsers[chatID] = make(map[uuid.UUID]bool)
		}
		if len(h.chatUsers[chatID]) == 0 {
			chatsToSubscribe = append(chatsToSubscribe, chatID)
		}
		h.chatUsers[chatID][userID] = true
	}

	h.mu.Unlock()

	if h.OnFirstChatUserJoined != nil {
		for _, chatID := range chatsToSubscribe {
			h.OnFirstChatUserJoined(chatID)
		}
	}

	return client
}

func (h *Hub) Unregister(client *Client) {
	h.mu.Lock()

	if c, ok := h.clients[client.userID]; ok && client == c {
		close(c.send)
		delete(h.clients, client.userID)
	} else {
		h.mu.Unlock()
		return
	}

	chatsToUnsubscribe := make([]uuid.UUID, 0)
	if chats, ok := h.userChats[client.userID]; ok {
		for chatID := range chats {
			if users, ok := h.chatUsers[chatID]; ok {
				delete(users, client.userID)
				if len(users) == 0 {
					delete(h.chatUsers, chatID)
					chatsToUnsubscribe = append(chatsToUnsubscribe, chatID)
				}
			}
		}
		delete(h.userChats, client.userID)
	}

	h.mu.Unlock()

	if h.OnLastChatUserLeft != nil {
		for _, chatID := range chatsToUnsubscribe {
			h.OnLastChatUserLeft(chatID)
		}
	}
}

func (h *Hub) BroadcastToChat(chatID uuid.UUID, payload any, excludeUserID uuid.UUID) {
	h.mu.RLock()
	userSet, exists := h.chatUsers[chatID]
	if !exists || len(userSet) == 0 {
		h.mu.RUnlock()
		return
	}

	targets := make([]*Client, 0, len(userSet))
	for uID := range userSet {
		if uID == excludeUserID {
			continue
		}
		if client, ok := h.clients[uID]; ok {
			targets = append(targets, client)
		}
	}
	h.mu.RUnlock()

	data, err := json.Marshal(payload)
	if err != nil {
		slog.Error("failed to marshal ws payload", slog.String("error", err.Error()))
		return
	}

	for _, client := range targets {
		select {
		case client.send <- data:
		default:
			slog.Warn("client send buffer full, dropping connection", slog.String("user_id", client.userID.String()))
			h.Unregister(client)
		}
	}
}
```

```
package pubsub

import (
	"context"
	"encoding/json"
	"fmt"
	"log/slog"
	"sync"

	"messenger/pkg/pubsub"
	"ws-gateway/internal/hub"

	"github.com/google/uuid"
)

type DynamicPubSubManager struct {
	ps       *pubsub.PubSub
	hub      *hub.Hub
	mu       sync.Mutex
	cancelFn map[uuid.UUID]context.CancelFunc
}

func NewDynamicPubSubManager(ps *pubsub.PubSub) *DynamicPubSubManager {
	return &DynamicPubSubManager{
		ps:       ps,
		cancelFn: make(map[uuid.UUID]context.CancelFunc),
	}
}

func (m *DynamicPubSubManager) SetHub(h *hub.Hub) {
	m.hub = h
}

func (m *DynamicPubSubManager) SubscribeChat(chatID uuid.UUID) {
	m.mu.Lock()
	defer m.mu.Unlock()

	if _, exists := m.cancelFn[chatID]; exists {
		return
	}

	ctx, cancel := context.WithCancel(context.Background())
	m.cancelFn[chatID] = cancel

	channel := fmt.Sprintf("chat:%s", chatID.String())

	go func() {
		err := m.ps.Subscribe(ctx, channel, func(payload []byte) {
			var event pubsub.ChatEvent
			if err := json.Unmarshal(payload, &event); err != nil {
				slog.Error("failed to unmarshal chat event", slog.String("error", err.Error()))
				return
			}

			senderID, _ := uuid.Parse(event.SenderID)
			if m.hub != nil {
				m.hub.BroadcastToChat(chatID, event, senderID)
			}
		})

		if err != nil && ctx.Err() == nil {
			slog.Error("pubsub subscription error", slog.String("channel", channel), slog.String("error", err.Error()))
		}
	}()
}

func (m *DynamicPubSubManager) UnsubscribeChat(chatID uuid.UUID) {
	m.mu.Lock()
	defer m.mu.Unlock()

	if cancel, exists := m.cancelFn[chatID]; exists {
		cancel()
		delete(m.cancelFn, chatID)
	}
}
```

```
const BASE_URL = '/api/v1';

class ApiError extends Error {
  status: number;

    constructor(status: number, message: string) {
      super(message);
      this.status = status;
    }
}

async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
    const res = await fetch(`${BASE_URL}${path}`, {
      ...options,
      credentials: 'include',
      headers: {
        'Content-Type': 'application/json',
        ...options.headers,
      },
    });

    if (!res.ok) {
      throw new ApiError(res.status, `Request failed: ${res.status}`);
    }

    if (res.status === 204) return undefined as T
    const text = await res.text()
    return (text ? JSON.parse(text) : undefined) as T
}
```

```
import { createContext, useCallback, useContext, useEffect, useMemo, useRef, type ReactNode } from "react";
import type { ChatEvent } from "../api/types";

type Handler = (event: ChatEvent) => void;

interface OutgoingEvent {
    type: string;
    chat_id: string;
    payload?: unknown;
}

interface WebSocketContextValue {
    subscribe: (handler: Handler) => () => void;
    send: (event: OutgoingEvent) => void;
}

const WebSocketContext = createContext<WebSocketContextValue | null>(null);

export function WebSocketProvider({ chatIds, children }: { chatIds: string[]; children: ReactNode }) {
    const chatIdsKey = chatIds.join(",");
    const socketRef = useRef<WebSocket | null>(null);
    const handlersRef = useRef<Set<Handler>>(new Set());

    useEffect(() => {
        if (!chatIdsKey) return;
        let closedByUs = false;
        let retryTimer: number | undefined;

        function connect() {
            const protocol = location.protocol === "https:" ? "wss" : "ws";
            const ws = new WebSocket(`${protocol}://${location.host}/api/v1/ws?chat_ids=${chatIdsKey}`);
            socketRef.current = ws;

            ws.onmessage = (e) => {
                const event: ChatEvent = JSON.parse(e.data);
                handlersRef.current.forEach((handler) => handler(event));
            };

            ws.onclose = () => {
                if (closedByUs) return;
                retryTimer = window.setTimeout(connect, 2000);
            };
            
            
        }

        connect();
        
        return () => {
            closedByUs = true;
            clearTimeout(retryTimer);
            socketRef.current?.close();
        };
    }, [chatIdsKey]);

    const subscribe = useCallback((handler: Handler) => {
        handlersRef.current.add(handler);
        return () => {
            handlersRef.current.delete(handler);
        };
    }, []);

    const send = useCallback((event: OutgoingEvent) => {
        if (socketRef.current?.readyState === WebSocket.OPEN) {
            socketRef.current.send(JSON.stringify(event));
        }
    }, []);

    const value = useMemo(() => ({ subscribe, send }), [subscribe, send]);

    return <WebSocketContext.Provider value={value}>{children}</WebSocketContext.Provider>;
}

export function useSocket() {
    const ctx = useContext(WebSocketContext);
    if (!ctx) throw new Error("useSocket must be used within WebSocketProvider");
    return ctx;
}
```

```
import type { ChatListItem, Message } from "../api/types";

export function applyMessage(
    prev: ChatListItem[],
    msg: Message,
    isUnread: boolean,
): ChatListItem[] {
    const existing = prev.find((c) => c.chat_id === msg.chat_id);
    if (!existing) return prev;

    const updated: ChatListItem = {
        ...existing,
        msg_text: msg.content,
        msg_sent_at: msg.created_at,
        msg_updated_at: msg.updated_at,
        unread_msg_cnt: existing.unread_msg_cnt + (isUnread ? 1 : 0),
    };

    return [updated, ...prev.filter((c) => c.chat_id !== msg.chat_id)];
}
```

```
package pubsub

import (
	"context"
	"encoding/json"
	"fmt"

	"github.com/redis/go-redis/v9"
)

type PubSub struct {
	client *redis.Client
}

func New(client *redis.Client) *PubSub {
	return &PubSub{client: client}
}

// Publish сериализует событие в JSON и отправляет в канал Redis
func (p *PubSub) Publish(ctx context.Context, channel string, event ChatEvent) error {
	data, err := json.Marshal(event)
	if err != nil {
		return fmt.Errorf("failed to marshal event: %w", err)
	}

	return p.client.Publish(ctx, channel, data).Err()
}

// Subscribe подписывается на канал и вызывает handler для каждого полученного сообщения
func (p *PubSub) Subscribe(ctx context.Context, channel string, handler func(payload []byte)) error {
	pubsub := p.client.Subscribe(ctx, channel)
	defer pubsub.Close()

	// Подтверждение установления подписки
	if _, err := pubsub.Receive(ctx); err != nil {
		return fmt.Errorf("subscribe failed for channel %s: %w", channel, err)
	}

	ch := pubsub.Channel()
	for {
		select {
		case <-ctx.Done():
			return ctx.Err()
		case msg, ok := <-ch:
			if !ok {
				return nil
			}
			handler([]byte(msg.Payload))
		}
	}
}
```
